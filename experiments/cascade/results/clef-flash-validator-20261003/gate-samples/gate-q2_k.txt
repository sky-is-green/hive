def fibonacci(n):

    """
    Generates Fibonacci sequence up to n terms.
    """
    a, b = 0, 1
    result = []
    for _ in range(n):
        result.append(a)
        a, b = b, a + b
    return result

print(fibonacci(5))

def s_is_square(s):
    """
    Check if s is a perfect square.
    """
    s_root = s ** 0.5
    return s_root.is_integer()

print(s_is_square(4))
